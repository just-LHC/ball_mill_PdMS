def plot_two_sensors(df, sensor_col="sensor_position", time_col="timestamp",
                     sensor_ids=None, features=None, normalize=False,
                     save_dir="sensor_plots", show=False):
    """
    Renders 2 distinct graphs (one per sensor), with all measured features 
    overlaid on the same graph with a full legend.

    Parameters:
    - df: DataFrame containing the dataset
    - sensor_col: Column identifying the sensors (e.g., 'sensor_position')
    - time_col: Column containing datetimes
    - sensor_ids: List of exactly 2 sensor IDs to plot (if None, picks the first 2 available)
    - features: List of feature names to overlay (if None, uses valid FEATURE_MAPPING columns)
    - normalize: If True, min-max scales features (0 to 1) so features with different units 
                 (e.g., Temperature 40°C vs Acceleration 2g) can be compared on the same scale
    - save_dir: Directory where the 2 graph files are saved
    - show: If True, renders figures interactively
    """
    if save_dir:
        os.makedirs(save_dir, exist_ok=True)

    available_sensors = df[sensor_col].dropna().unique().tolist()
    if len(available_sensors) < 2:
        raise ValueError(f"Dataset must contain at least 2 distinct sensors, found: {available_sensors}")

    # Default to first two sensors if not explicitly passed
    if sensor_ids is None:
        sensor_ids = available_sensors[:2]
    elif len(sensor_ids) != 2:
        raise ValueError(f"Expected exactly 2 sensors, got {len(sensor_ids)}: {sensor_ids}")

    # Target only legitimate numerical sensor features
    if features is None:
        features = [name for name in FEATURE_MAPPING.values() if name in df.columns]

    print(f"\n[Plotting] Generating 2 separate graphs for sensors: {sensor_ids}...")

    saved_paths = []
    for sensor in sensor_ids:
        sensor_data = df[df[sensor_col] == sensor].sort_values(by=time_col).copy()
        if sensor_data.empty:
            print(f"   -> Warning: No records found for sensor '{sensor}'")
            continue

        fig, ax = plt.subplots(figsize=(13, 6))
        fig.patch.set_facecolor("white")

        for feat in features:
            if feat not in sensor_data.columns:
                continue
            y = sensor_data[feat]
            
            if normalize:
                min_v, max_v = y.min(), y.max()
                y = (y - min_v) / (max_v - min_v) if max_v > min_v else y * 0
                label_name = f"{feat} (normalized)"
            else:
                label_name = feat

            ax.plot(sensor_data[time_col], y, label=label_name, linewidth=1.4, alpha=0.85)

        ylabel = "Normalized Scale (0-1)" if normalize else "Measured Engineering Units"
        ax.set_ylabel(ylabel, fontsize=11, fontweight="semibold")
        ax.set_xlabel("Acquisition Timestamp", fontsize=11, fontweight="semibold")
        ax.set_title(f"All Features: Sensor {sensor}", fontsize=13, fontweight="bold", pad=12)
        ax.grid(True, linestyle="--", alpha=0.5)
        ax.legend(loc="upper right", framealpha=0.9, fontsize=9)

        plt.tight_layout()

        if save_dir:
            safe_name = str(sensor).replace("/", "_").replace("\\", "_")
            out_file = os.path.join(save_dir, f"sensor_graph_{safe_name}.png")
            plt.savefig(out_file, dpi=150)
            saved_paths.append(out_file)
            print(f"   -> Saved: {out_file}")

        if show:
            plt.show()

        plt.close(fig)

    return saved_paths

try:

        "Content-Type": "application/json"
    })

    # 2. Date Range Verification[cite: 3]
    if not START_DATE or not END_DATE:
        now = datetime.now()
        END_DATE = now.strftime("%Y-%m-%d %H:%M:%S")
        START_DATE = (now - timedelta(days=7)).strftime("%Y-%m-%d %H:%M:%S")

    print(f"\n[2/4] Fetching NV measurements from {START_DATE} to {END_DATE}...")
    all_rows = []

    # 3. Retrieve NV measurements per asset[cite: 3, 4]
    for asset_id in ASSET_IDS:
        url = f"{BASE_URL}/tenant/assets/{asset_id}/nv-measurements"
        params = {"startDate": START_DATE, "endDate": END_DATE}
        
        response = session.get(url, params=params, timeout=30)
        if response.status_code != 200:
            print(f"   -> Asset {asset_id}: Failed (Status {response.status_code})")
            continue

        res_json = response.json()
        data = res_json.get("data", {})
        if not data:
            print(f"   -> Asset {asset_id}: No data points returned.")
            continue

        for timestamp, sensor_dict in data.items():
            if not isinstance(sensor_dict, dict):
                continue
            for sensor_pos, features in sensor_dict.items():
                if not isinstance(features, dict):
                    continue
                row = {
                    "timestamp": timestamp,
                    "asset_id": asset_id,
                    "sensor_position": sensor_pos
                }
                for fid_str, val in features.items():
                    fid = int(fid_str) if str(fid_str).isdigit() else fid_str
                    col_name = FEATURE_MAPPING.get(fid, f"Feature_{fid}")
                    row[col_name] = val
                all_rows.append(row)
                
        print(f"   -> Asset {asset_id}: Successfully processed.")

    if not all_rows:
        raise ValueError("No measurement data could be retrieved. Verify date range and asset IDs.")

    # 4. Assemble ML Dataset[cite: 1]
    print("\n[3/4] Generating final structured dataset...")
    df_final = pd.DataFrame(all_rows)
    df_final["timestamp"] = pd.to_datetime(df_final["timestamp"])
    df_final = df_final.sort_values(by=["timestamp", "sensor_position"])

    output_name = "ball_mill_ml_dataset.csv"
    df_final.to_csv(output_name, index=False)
    print(f"Saved dataset to '{output_name}' ({len(df_final)} rows, {df_final['sensor_position'].nunique()} sensors)")

    # 5. Plot 2 Sensors Across All Features
    print("\n[4/4] Executing Sensor Visualization...")
    plot_two_sensors(
        df=df_final,
        sensor_col="sensor_position",
        time_col="timestamp",

        normalize=False,         # Set to True if you want 0-1 min-max scaling to align diverse units
        save_dir="two_graphs_output",
        show=False
